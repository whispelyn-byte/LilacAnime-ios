cmake_minimum_required(VERSION 3.22.1)
project(lilac_whisper CXX C)

# whisper.cpp v1.7.4. CI clones it into androidWhisper/build/whisper.cpp first; a local build fetches it here.
set(WHISPER_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/whisper.cpp")
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_CURL OFF CACHE BOOL "" FORCE)
set(WHISPER_SDL2 OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
if (EXISTS "${WHISPER_DIR}/CMakeLists.txt")
    add_subdirectory("${WHISPER_DIR}" "${CMAKE_BINARY_DIR}/whisper")
else()
    include(FetchContent)
    FetchContent_Declare(whisper GIT_REPOSITORY https://github.com/ggml-org/whisper.cpp.git GIT_TAG v1.7.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(whisper)
endif()

add_library(lilac_whisper SHARED whisper_jni.cpp)
set_target_properties(lilac_whisper PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON)
target_compile_options(lilac_whisper PRIVATE -O3)
# The static ggml inside stays private to this library (llama.cpp's own ggml lives in liblilac_local_ai.so).
target_link_options(lilac_whisper PRIVATE "-Wl,--exclude-libs,ALL")
target_link_libraries(lilac_whisper PRIVATE whisper android log)
